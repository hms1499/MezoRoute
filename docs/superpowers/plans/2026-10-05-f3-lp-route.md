# F3 LP Route Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A user with wallet MUSD deposits it into the MUSD/BTC LP through `MezoRouteExecutor.enter`: route page with an exact preview, confirm page with a permit (EOA) or exact approval (smart account) stepper, and an event-derived receipt page.

**Architecture:**

- **Pure core, unit-tested:** `lib/quote/entry` (optimal swap + exact integer simulation of Pool/Router), `lib/quote/gas`, and `lib/routes/musd-btc-lp/{snapshot,intent,steps,gate,after-entry,preview,permit,params,receipt}`, plus `lib/tx/{send,stepper}` and the error decoder.
- **Reads:** one multicall snapshot per page (F2's dashboard reads + the LP-route reads), refetched every 15 s. The URL carries only `amount` and `slippage`; every page re-quotes from the chain.
- **UI:** thin client components in `components/lp/` and `components/receipt/`, behind `next/dynamic` pages at `/route/lp/`, `/route/lp/confirm/`, `/tx/`. `useEntryFlow` runs the stepper: permit (`signTypedData`) or approval (`sendCall`), then re-quote → simulate → `submitCall` → navigate to the receipt.

**Tech Stack:** Next.js 14.2.35 (static export, `trailingSlash: true`), React 18.3.1, `@mezo-org/passport` 0.17.2, RainbowKit 2.0.2, wagmi 2.19.5, viem 2.57.2, TanStack Query 5, Tailwind 3, Vitest 5, TypeScript 5.9.

**Spec:** `docs/superpowers/specs/2026-10-05-f3-lp-route-design.md`. Read it first: it holds the chain facts verified on 5 Oct 2026 (MUSD permit domain, Tigris fee handling, S1 replays). Product context: `PRODUCT_SPEC.md` §7 Flow C, §9 screens 2a/3a/4, §11.2–11.3, §12, §13.

```ts
import { ceilDiv, isqrt, WAD } from "@/lib/math";

const BPS = 10_000n;

/** The MUSD/BTC pool as the quote sees it: reserves by token (not token0/token1), LP supply, swap fee in bps. */
export type PoolState = { reserveMusd: bigint; reserveBtc: bigint; totalSupply: bigint; feeBps: bigint };

/** What `enter` does with `net` MUSD (after MezoRoute's fee) and a given swap amount. */
export type EntrySimulation = {
  musdToSwap: bigint;
  btcFromSwap: bigint;
  musdAdded: bigint;
  btcAdded: bigint;
  liquidity: bigint;
  musdRefund: bigint;
  btcRefund: bigint;
  after: PoolState; // the pool after the swap and the deposit
};

export type EntryQuote = EntrySimulation & {
  musdIn: bigint;
  fee: bigint; // MezoRoute's execution fee
  net: bigint; // musdIn − fee
  priceImpact: bigint; // 1e18 = 100%, against the pre-trade mid price, rounded up
};

export type EntryMinimums = { minBtcFromSwap: bigint; minMusdAdded: bigint; minBtcAdded: bigint; minLpOut: bigint };

/** Slippage presets in bps (PRODUCT_SPEC §12); never raised silently. */
export const SLIPPAGE_PRESETS = [50n, 100n, 200n] as const;
export type SlippageBps = (typeof SLIPPAGE_PRESETS)[number];
export const DEFAULT_SLIPPAGE: SlippageBps = 100n;

/**
 * MUSD to swap so that the BTC received and the MUSD kept match the post-swap reserve ratio. Tigris
 * moves the swap fee out of the pool (Pool._update0), so with g = 1 − f the condition is
 * g²s² + r(2 − f)s − a·r = 0 (PRODUCT_SPEC §12), solved here in bps-scaled integers.
 */
export function optimalSwap(net: bigint, reserveMusd: bigint, feeBps: bigint): bigint {
  if (net <= 0n || reserveMusd <= 0n) return 0n;
  const g = BPS - feeBps;
  const b = reserveMusd * (2n * BPS - feeBps);
  return ((isqrt(b * b + 4n * g * g * net * reserveMusd) - b) * BPS) / (2n * g * g);
}

/**
 * The executor's entry, step by step with the contracts' rounding: Pool.swap takes the fee out of
 * the pool and prices the rest with getAmountOut; Router._addLiquidity sizes the deposit with
 * quoteLiquidity against the post-swap reserves; Pool.mint gives the smaller of the two shares.
 */
export function simulateEntry(pool: PoolState, net: bigint, musdToSwap: bigint): EntrySimulation {
  const { reserveMusd, reserveBtc, totalSupply, feeBps } = pool;
  const swapIn = musdToSwap - (musdToSwap * feeBps) / BPS;
  const btcFromSwap = (swapIn * reserveBtc) / (reserveMusd + swapIn);
  const musdAfterSwap = reserveMusd + swapIn;
  const btcAfterSwap = reserveBtc - btcFromSwap;

  const musdDesired = net - musdToSwap;
  const btcOptimal = (musdDesired * btcAfterSwap) / musdAfterSwap;
  const [musdAdded, btcAdded] =
    btcOptimal <= btcFromSwap
      ? [musdDesired, btcOptimal]
      : [(btcFromSwap * musdAfterSwap) / btcAfterSwap, btcFromSwap];
  const byMusd = (musdAdded * totalSupply) / musdAfterSwap;
  const byBtc = (btcAdded * totalSupply) / btcAfterSwap;
  const liquidity = byMusd < byBtc ? byMusd : byBtc;

  return {
    musdToSwap,
    btcFromSwap,
    musdAdded,
    btcAdded,
    liquidity,
    musdRefund: musdDesired - musdAdded,
    btcRefund: btcFromSwap - btcAdded,
    after: {
      reserveMusd: musdAfterSwap + musdAdded,
      reserveBtc: btcAfterSwap + btcAdded,
      totalSupply: totalSupply + liquidity,
      feeBps,
    },
  };
}

/** 1 − btcFromSwap × r / (s × R): the loss against the pre-trade mid price, including the pool fee. */
export function priceImpact(pool: PoolState, musdToSwap: bigint, btcFromSwap: bigint): bigint {
  if (musdToSwap === 0n) return 0n;
  const atMid = musdToSwap * pool.reserveBtc;
  const received = btcFromSwap * pool.reserveMusd;
  return ceilDiv((atMid - received) * WAD, atMid);
}

/** The full entry quote for `musdIn`; null without an amount or with an empty pool. */
export function quoteEntry(pool: PoolState, musdIn: bigint, executorFeeBps: bigint): EntryQuote | null {
  if (musdIn <= 0n || pool.reserveMusd === 0n || pool.reserveBtc === 0n || pool.totalSupply === 0n) return null;
  // The executor charges its fee before splitting the input (spec 11.3 step 5).
  const fee = (musdIn * executorFeeBps) / BPS;
  const net = musdIn - fee;
  const simulation = simulateEntry(pool, net, optimalSwap(net, pool.reserveMusd, pool.feeBps));
  return {
    ...simulation,
    musdIn,
    fee,
    net,
    priceImpact: priceImpact(pool, simulation.musdToSwap, simulation.btcFromSwap),
  };
}

/** Each expected value minus the slippage, rounded down (the four minimums of EnterParams). */
export function entryMinimums(simulation: EntrySimulation, slippageBps: bigint): EntryMinimums {
  const floor = (value: bigint) => (value * (BPS - slippageBps)) / BPS;
  return {
    minBtcFromSwap: floor(simulation.btcFromSwap),
    minMusdAdded: floor(simulation.musdAdded),
    minBtcAdded: floor(simulation.btcAdded),
    minLpOut: floor(simulation.liquidity),
  };
}
```

```ts
/** Everything the LP route reads, from one block. */
export type LpRouteSnapshot = {
  dashboard: DashboardSnapshot;
  pool: PoolState;
  allowance: bigint; // MUSD the executor may already pull from the account
  permitNonce: bigint; // MUSD EIP-2612 nonce of the account
  executorFeeBps: bigint; // read on chain; the gate compares it with the pinned fee
  maxMusdIn: bigint; // the executor's per-transaction entry cap
  blockNumber: bigint;
  blockTimestamp: bigint;

/** How many reads `snapshotContracts` makes; the LP-route reads follow them. */
export const DASHBOARD_READS = 12;

type LpRouteTail = readonly [
  poolFeeBps: bigint,
  allowance: bigint,
  permitNonce: bigint,
  executorFeeBps: bigint,
  maxMusdIn: bigint,
  blockNumber: bigint,
  blockTimestamp: bigint,
];
export type LpRouteResults = readonly [...SnapshotResults, ...LpRouteTail];

/** The dashboard's reads plus the LP route's, one multicall at one block, all on the selected network. */
export function lpRouteContracts(network: NetworkConfig, executor: Address, account: Address) {
  const { addresses: a, chainId } = network;
  return [
    ...snapshotContracts(network, account),
    { chainId, address: a.poolFactory, abi: poolFactoryAbi, functionName: "getFee", args: [a.pool, false] },
    { chainId, address: a.musd, abi: erc20Abi, functionName: "allowance", args: [account, executor] },
    { chainId, address: a.musd, abi: musdPermitAbi, functionName: "nonces", args: [account] },
    { chainId, address: executor, abi: mezoRouteExecutorAbi, functionName: "feeBps" },
    { chainId, address: executor, abi: mezoRouteExecutorAbi, functionName: "maxMusdIn" },
    { chainId, address: a.multicall3, abi: multicall3Abi, functionName: "getBlockNumber" },
    { chainId, address: a.multicall3, abi: multicall3Abi, functionName: "getCurrentBlockTimestamp" },
  ] as const;
}

export function parseLpRouteSnapshot(network: NetworkConfig, results: LpRouteResults): LpRouteSnapshot {
  const dashboard = parseSnapshot(network, results.slice(0, DASHBOARD_READS) as unknown as SnapshotResults);
  const [poolFeeBps, allowance, permitNonce, executorFeeBps, maxMusdIn, blockNumber, blockTimestamp] = results.slice(
    DASHBOARD_READS,
  ) as unknown as LpRouteTail;
  return {
    dashboard,
    pool: {
      reserveMusd: dashboard.lp.reserveMusd,
      reserveBtc: dashboard.lp.reserveBtc,
      totalSupply: dashboard.lp.totalSupply,
      feeBps: poolFeeBps,
    },
    allowance,
    permitNonce,
    executorFeeBps,
    maxMusdIn,
    blockNumber,
    blockTimestamp,
  };
}
```

```ts
import type { NetworkConfig } from "@/lib/config/networks";
import { writeBlocker } from "@/lib/config/write-gate";
import {
  aboveCap,
  AMOUNT_TOO_SMALL,
  FEE_MISMATCH,
  INSUFFICIENT_BALANCE,
  needsGas,
  noExecutor,
  PRICE_IMPACT,
  QUOTE_EXPIRED,
  wrongNetwork,
} from "@/lib/errors/outcomes";
import type { Outcome } from "@/lib/errors/types";
import type { EntryQuote } from "@/lib/quote/entry";

/** 5% (PRODUCT_SPEC §12, provisional until integration tests); 1e18 = 100%. */
export const PRICE_IMPACT_LIMIT = 50_000_000_000_000_000n;
/** A quote's lifetime (PRODUCT_SPEC §12). */
export const QUOTE_TTL_MS = 60_000;

export type EntryGateInput = {
  network: NetworkConfig;
  walletChainId: number | undefined;
  chainFeeBps: bigint;
  musdIn: bigint | null;
  walletMusd: bigint;
  maxMusdIn: bigint;
  quote: EntryQuote | null;
  lacksGas: boolean;
  quoteAgeMs: number;
};

export type EntryBlockReason =
  | "no-executor"
  | "fee-mismatch"
  | "wrong-network"
  | "zero"
  | "insufficient-balance"
  | "above-cap"
  | "too-small"
  | "price-impact"
  | "needs-gas"
  | "quote-expired";

/** `outcome` is null when the button simply stays disabled (no amount yet). */
export type EntryBlock = { reason: EntryBlockReason; outcome: Outcome | null };

/** Why entry is blocked; the first matching reason wins (FR-01, FR-05, FR-22). */
export function entryBlock(input: EntryGateInput): EntryBlock | null {
  const { network, quote } = input;
  const block = (reason: EntryBlockReason, outcome: Outcome | null): EntryBlock => ({ reason, outcome });
  if (!network.executor) return block("no-executor", noExecutor(network));
  if (input.chainFeeBps !== network.executor.feeBps) return block("fee-mismatch", FEE_MISMATCH);
  if (writeBlocker(network, input.walletChainId) === "wrong-network") return block("wrong-network", wrongNetwork(network));
  if (input.musdIn === null) return block("zero", null);
  if (input.musdIn > input.walletMusd) return block("insufficient-balance", INSUFFICIENT_BALANCE);
  if (input.musdIn > input.maxMusdIn) return block("above-cap", aboveCap(network));
  if (
    !quote ||
    quote.musdToSwap === 0n ||
    quote.musdToSwap >= quote.net ||
    quote.btcFromSwap === 0n ||
    quote.liquidity === 0n
  ) {
    return block("too-small", AMOUNT_TOO_SMALL);
  }
  if (quote.priceImpact > PRICE_IMPACT_LIMIT) return block("price-impact", PRICE_IMPACT);
  if (input.lacksGas) return block("needs-gas", needsGas(network));
  if (input.quoteAgeMs > QUOTE_TTL_MS) return block("quote-expired", QUOTE_EXPIRED);
  return null;
}
```

```ts
import { parseMusdAmount } from "@/lib/quote/amount";
import { DEFAULT_SLIPPAGE, type SlippageBps } from "@/lib/quote/entry";

/** What the user asked for; the only state the LP pages pass between them (in the URL). */
export type EntryIntent = { amount: string; musdIn: bigint | null; slippageBps: SlippageBps };

const SLIPPAGE_BY_PARAM: Record<string, SlippageBps> = { "0.5": 50n, "1": 100n, "2": 200n };
const PARAM_BY_SLIPPAGE: Record<string, string> = { "50": "0.5", "100": "1", "200": "2" };

export function parseEntryIntent(params: { get(name: string): string | null }): EntryIntent {
  const amount = params.get("amount") ?? "";
  const slippage = params.get("slippage") ?? "";
  return {
    amount,
    musdIn: parseMusdAmount(amount),
    slippageBps: Object.hasOwn(SLIPPAGE_BY_PARAM, slippage) ? SLIPPAGE_BY_PARAM[slippage] : DEFAULT_SLIPPAGE,
  };
}

/** The query both LP pages share: the user's inputs only, never quote values. */
export function entryQuery(intent: { amount: string; slippageBps: SlippageBps }): string {
  const query = new URLSearchParams();
  if (intent.amount) query.set("amount", intent.amount);
  query.set("slippage", PARAM_BY_SLIPPAGE[intent.slippageBps.toString()]);
  return `?${query.toString()}`;
}
```

```ts
import type { NetworkConfig } from "@/lib/config/networks";
import type { ExecutorPermit } from "./params";

/**
 * MUSD's EIP-712 domain, from `eip712Domain()` on testnet (5 Oct 2026): name "Mezo USD", version
 * "1". Check mainnet's `eip712Domain()` before M1 enables mainnet writes.
 */
export function musdPermitDomain(network: NetworkConfig) {
  return {
    name: "Mezo USD",
    version: "1",
    chainId: network.chainId,
    verifyingContract: network.addresses.musd,
  } as const;

export const PERMIT_TYPES = {
  Permit: [
    { name: "owner", type: "address" },
    { name: "spender", type: "address" },
    { name: "value", type: "uint256" },
    { name: "nonce", type: "uint256" },
    { name: "deadline", type: "uint256" },
  ],
} as const;

/** EIP-2612 permit for exactly `value` MUSD to the executor (FR-06). */
export function permitTypedData(input: {
  network: NetworkConfig;
  owner: Address;
  spender: Address;
  value: bigint;
  nonce: bigint;
  deadline: bigint;
}) {
  return {
    domain: musdPermitDomain(input.network),
    types: PERMIT_TYPES,
    primaryType: "Permit" as const,
    message: {
      owner: input.owner,
      spender: input.spender,
      value: input.value,
      nonce: input.nonce,
      deadline: input.deadline,
    },
  };
}

export function toExecutorPermit(signature: Hex, value: bigint, deadline: bigint): ExecutorPermit {
  const { r, s, v, yParity } = parseSignature(signature);
  return { value, deadline, v: Number(v ?? BigInt(yParity + 27)), r, s };
}

/** A permit is sent only with a minute to spare, so it cannot expire while the transaction is pending. */
export const PERMIT_MARGIN_SECONDS = 60n;

export function permitUsable(permit: ExecutorPermit, nowSeconds: bigint): boolean {
  return nowSeconds + PERMIT_MARGIN_SECONDS < permit.deadline;
}
```

```ts
import type { Hash } from "viem";
import type { DecodedError } from "@/lib/errors/types";
import type { StepKind } from "@/lib/routes/musd-btc-lp/steps";

export type StepStatus = "idle" | "awaiting-wallet" | "submitted" | "done" | "failed";
export type Step = { kind: StepKind; status: StepStatus; hash?: Hash; error?: DecodedError };
/** `active` is the index of the step the user acts on next; it equals `steps.length` when all are done. */
export type Stepper = { steps: Step[]; active: number };

export type StepperEvent =
  | { type: "start" }
  | { type: "signed" }
  | { type: "submitted"; hash: Hash }
  | { type: "confirmed" }
  | { type: "failed"; error: DecodedError }
  | { type: "permit-expired" }
  | { type: "reset"; kinds: StepKind[] };

export function createStepper(kinds: StepKind[]): Stepper {
  return { steps: kinds.map((kind) => ({ kind, status: "idle" })), active: 0 };
}

export function hasStarted(state: Stepper): boolean {
  return state.steps.some((step) => step.status !== "idle");
}

/** The confirm page's stepper (spec §9 screen 3a); events that do not fit the active step are ignored. */
export function stepperReducer(state: Stepper, event: StepperEvent): Stepper {
  const step = state.steps[state.active];
  const replace = (next: Step, advance = false): Stepper => ({
    steps: state.steps.map((current, index) => (index === state.active ? next : current)),
    active: advance ? state.active + 1 : state.active,
  });

  switch (event.type) {
    case "reset":
      return hasStarted(state) ? state : createStepper(event.kinds);
    case "permit-expired": {
      const permit = state.steps.findIndex((current) => current.kind === "permit");
      if (permit === -1) return state;
      return {
        steps: state.steps.map((current, index) => (index >= permit ? { kind: current.kind, status: "idle" } : current)),
        active: permit,
      };
    }
    case "start":
      return step && (step.status === "idle" || step.status === "failed")
        ? replace({ kind: step.kind, status: "awaiting-wallet" })
        : state;
    case "signed":
      return step?.kind === "permit" && step.status === "awaiting-wallet" ? replace({ ...step, status: "done" }, true) : state;
    case "submitted":
      return step && step.kind !== "permit" && step.status === "awaiting-wallet"
        ? replace({ ...step, status: "submitted", hash: event.hash })
        : state;
    case "confirmed":
      return step?.status === "submitted" ? replace({ ...step, status: "done" }, true) : state;
    case "failed":
      return step && (step.status === "awaiting-wallet" || step.status === "submitted")
        ? replace({ ...step, status: "failed", error: event.error })
        : state;
  }
}
```

```ts
"use client";

import type { Address } from "viem";
import { useBalance, useGasPrice, useReadContracts } from "wagmi";
import { useNetwork } from "@/lib/config/network-context";
import { lpRouteContracts, parseLpRouteSnapshot, type LpRouteData, type LpRouteResults } from "./snapshot";

/** New quote every 15 s; a quote older than 60 s is blocked (gate). */
export const QUOTE_REFRESH_MS = 15_000;

/**
 * The LP route's reads on the selected network: one multicall snapshot, plus gas balance and price.
 * `refresh` refetches all three and resolves with the fresh data (the send step re-quotes from it).
 */
export function useLpRouteSnapshot(
  account: Address,
  executor: Address,
): { data: LpRouteData | undefined; error: Error | null; refresh: () => Promise<LpRouteData> } {
  const { network } = useNetwork();
  const reads = useReadContracts({
    contracts: lpRouteContracts(network, executor, account),
    allowFailure: false,
    query: { refetchInterval: QUOTE_REFRESH_MS },
  });
  const gas = useBalance({ address: account, chainId: network.chainId, query: { refetchInterval: QUOTE_REFRESH_MS } });
  const gasPrice = useGasPrice({ chainId: network.chainId, query: { refetchInterval: QUOTE_REFRESH_MS } });

  const toData = (results: unknown, gasBalance: bigint, price: bigint | undefined, fetchedAt: number): LpRouteData => ({
    snapshot: parseLpRouteSnapshot(network, results as LpRouteResults),
    gasBalance,
    gasPrice: price,
    fetchedAt,
  });

  return {
    data: reads.data && gas.data ? toData(reads.data, gas.data.value, gasPrice.data, reads.dataUpdatedAt) : undefined,
    error: reads.error ?? gas.error ?? null,
    refresh: async () => {
      const [freshReads, freshGas, freshPrice] = await Promise.all([reads.refetch(), gas.refetch(), gasPrice.refetch()]);
      if (freshReads.error) throw freshReads.error;
      if (freshGas.error) throw freshGas.error;
      if (!freshReads.data || !freshGas.data) throw new Error("LP route snapshot unavailable");
      return toData(freshReads.data, freshGas.data.value, freshPrice.data, freshReads.dataUpdatedAt);
    },
  };
}
```

```ts
import { useEffect, useState } from "react";

/** The current time in ms, re-rendering every `intervalMs` (quote countdowns). */
export function useNow(intervalMs: number): number {
  const [now, setNow] = useState(() => Date.now());
  useEffect(() => {
    const id = setInterval(() => setNow(Date.now()), intervalMs);
    return () => clearInterval(id);
  }, [intervalMs]);
  return now;
}
```

```ts
"use client";

import { useRouter } from "next/navigation";
import { useEffect, useReducer, useRef } from "react";
import type { Address } from "viem";
import { useConfig } from "wagmi";
import { getAccount, signTypedData, simulateContract } from "wagmi/actions";
import { useErrorToast } from "@/components/ErrorToast";
import { mezoRouteExecutorAbi } from "@/lib/abi/mezoRouteExecutor";
import { useNetwork } from "@/lib/config/network-context";
import { decodeError } from "@/lib/errors/decode";
import { AMOUNT_TOO_SMALL, PERMIT_EXPIRED } from "@/lib/errors/outcomes";
import { BlockedError } from "@/lib/tx/errors";
import { confirmCall, submitCall, type SendPin } from "@/lib/tx/send";
import { createStepper, hasStarted, stepperReducer, type Stepper } from "@/lib/tx/stepper";
import type { WalletKind } from "@/lib/wallet/capabilities";
import { entryQuery, type EntryIntent } from "./intent";
import { buildEnterParams, encodeApproveCall, encodeEnterCall, NO_PERMIT, DEADLINE_SECONDS, type ExecutorPermit } from "./params";
import { permitTypedData, permitUsable, toExecutorPermit } from "./permit";
import { entryPreview } from "./preview";
import type { LpRouteData } from "./snapshot";
import type { StepKind } from "./steps";

export type EntryFlowInput = {
  account: Address;
  executor: Address;
  intent: EntryIntent;
  walletKind: WalletKind;
  plan: StepKind[];
  refresh: () => Promise<LpRouteData>;
};

const nowSeconds = () => BigInt(Math.floor(Date.now() / 1000));

/**
 * Runs the confirm stepper: permit (EOA) or exact approval (smart account), then enter. The account
 * and chain are pinned on the first step; enter re-quotes, simulates, submits, and hands the hash
 * to the receipt page.
 */
export function useEntryFlow(input: EntryFlowInput): { stepper: Stepper; run: () => void } {
  const config = useConfig();
  const { network } = useNetwork();
  const router = useRouter();
  const { showError } = useErrorToast();
  const [stepper, dispatch] = useReducer(stepperReducer, input.plan, createStepper);
  const stepperRef = useRef(stepper);
  stepperRef.current = stepper;
  const pin = useRef<SendPin | null>(null);
  const permit = useRef<ExecutorPermit | null>(null);
  const planKey = input.plan.join(",");

  useEffect(() => {
    if (!hasStarted(stepperRef.current)) dispatch({ type: "reset", kinds: planKey.split(",") as StepKind[] });
  }, [planKey]);

  const musdIn = input.intent.musdIn ?? 0n;

  async function signPermit(sendPin: SendPin) {
    const fresh = await input.refresh();
    const deadline = fresh.snapshot.blockTimestamp + DEADLINE_SECONDS;
    const typedData = permitTypedData({
      network,
      owner: sendPin.account,
      spender: input.executor,
      value: musdIn,
      nonce: fresh.snapshot.permitNonce,
      deadline,
    });
    const signature = await signTypedData(config, { account: sendPin.account, ...typedData });
    permit.current = toExecutorPermit(signature, musdIn, deadline);
    dispatch({ type: "signed" });
  }

  async function approve(sendPin: SendPin) {
    const hash = await submitCall(config, { to: network.addresses.musd, data: encodeApproveCall(input.executor, musdIn) }, sendPin);
    dispatch({ type: "submitted", hash });
    await confirmCall(config, hash, sendPin.chainId);
    dispatch({ type: "confirmed" });
  }

  async function enter(sendPin: SendPin) {
    const hasPermitStep = stepperRef.current.steps.some((step) => step.kind === "permit");
    const signed = permit.current;
    if (hasPermitStep && (!signed || !permitUsable(signed, nowSeconds()))) {
      permit.current = null;
      dispatch({ type: "permit-expired" });
      throw new BlockedError(PERMIT_EXPIRED);
    }
    const fresh = await input.refresh();
    const preview = entryPreview({
      network,
      data: fresh,
      intent: input.intent,
      walletKind: input.walletKind,
      walletChainId: getAccount(config).chainId,
      now: Date.now(),
    });
    if (preview.block) throw new BlockedError(preview.block.outcome ?? AMOUNT_TOO_SMALL);
    if (!preview.quote || !preview.minimums) throw new BlockedError(AMOUNT_TOO_SMALL);
    const params = buildEnterParams(preview.quote, preview.minimums, preview.deadline, sendPin.account);
    const usedPermit = hasPermitStep && signed ? signed : NO_PERMIT;
    // A simulated revert is decoded and nothing is sent.
    await simulateContract(config, {
      address: input.executor,
      abi: mezoRouteExecutorAbi,
      functionName: "enter",
      args: [params, usedPermit],
      account: sendPin.account,
      chainId: sendPin.chainId,
    });
    const hash = await submitCall(config, { to: input.executor, data: encodeEnterCall(params, usedPermit) }, sendPin);
    dispatch({ type: "submitted", hash });
    router.push(`/tx/?hash=${hash}`);
  }

  async function runActive() {
    const step = stepperRef.current.steps[stepperRef.current.active];
    if (!step) return;
    pin.current ??= { account: input.account, chainId: network.chainId };
    const sendPin = pin.current;
    dispatch({ type: "start" });
    try {
      if (step.kind === "permit") await signPermit(sendPin);
      else if (step.kind === "approve") await approve(sendPin);
      else await enter(sendPin);
    } catch (error) {
      dispatch({ type: "failed", error: decodeError(error, network) });
      showError(error, {
        retry: () => void runActive(),
        refreshQuote: () => void input.refresh().catch(() => undefined),
        reduceAmount: () => router.push(`/route/lp/${entryQuery(input.intent)}`),
      });
    }
  }

  return { stepper, run: () => void runActive() };
}
```

```tsx
"use client";

import type { EntryBlock } from "@/lib/routes/musd-btc-lp/gate";
import type { StepKind } from "@/lib/routes/musd-btc-lp/steps";
import { useEntryFlow, type EntryFlowInput } from "@/lib/routes/musd-btc-lp/useEntryFlow";
import type { Step } from "@/lib/tx/stepper";
import { BlockNotice } from "./BlockNotice";

const COPY: Record<StepKind, { title: string; detail: string; action: string }> = {
  permit: {
    title: "Sign a MUSD permit",
    detail: "Lets MezoRoute take exactly this amount in the next step. Signing costs no gas.",
    action: "Sign",
  },
  approve: {
    title: "Approve MUSD",
    detail: "Your Bitcoin wallet approves the exact amount in a separate transaction.",
    action: "Approve",
  },
  enter: {
    title: "Deposit into the pool",
    detail: "One transaction: fee, swap, and add liquidity. Leftovers return to you.",
    action: "Send",
  },

export function EntryStepper({ block, ...flow }: EntryFlowInput & { block: EntryBlock | null }) {
  const { stepper, run } = useEntryFlow(flow);
  return (
    <section className="rounded-2xl bg-white p-5 shadow-card">
      <h2 className="font-semibold">Confirm</h2>
      <ol className="mt-3 space-y-3">
        {stepper.steps.map((step, index) => (
          <li key={step.kind} className="flex items-start justify-between gap-3 rounded-xl border border-line p-3">
            <div>
              <p className="text-sm font-semibold">
                {index + 1}. {COPY[step.kind].title}
              </p>
              <p className="text-xs text-muted">{COPY[step.kind].detail}</p>
              {step.status === "failed" && step.error && <p className="mt-1 text-xs text-danger">{step.error.message}</p>}
            </div>
            <StepAction step={step} active={index === stepper.active} disabled={block !== null} onRun={run} />
          </li>
        ))}
      </ol>

        <BlockNotice block={block} />
      </div>
    </section>
  );
}

function StepAction({ step, active, disabled, onRun }: { step: Step; active: boolean; disabled: boolean; onRun: () => void }) {
```

```tsx
/** Spec §9 screen 4: progress until confirmed, then values from the Entered event only (FR-11, FR-13). */

function Receipt({ hash }: { hash: Hash }) {
  const { network } = useNetwork();
  const config = useConfig();
  const queryClient = useQueryClient();
  const receipt = useQuery({
    queryKey: [RECEIPT_KEY, network.chainId, hash],
    queryFn: () => confirmCall(config, hash, network.chainId),
    retry: false,
    staleTime: Infinity,
    refetchOnWindowFocus: false,
  });

  // FR-02: balances refresh after a confirmed transaction.
  useEffect(() => {
    if (receipt.isSuccess) {
      void queryClient.invalidateQueries({ predicate: (query) => query.queryKey[0] !== RECEIPT_KEY });
    }
  }, [receipt.isSuccess, queryClient]);

  const explorer = explorerTxUrl(network, receipt.data?.transactionHash ?? hash);
  return (
    <Card>
      {receipt.isPending && (
        <>
          <h1 className="text-lg font-semibold">Confirming…</h1>
          <p className="mt-1 text-sm text-muted">Transaction {shortAddress(hash)} is submitted. This usually takes a few seconds.</p>
        </>
      )}
      {receipt.isError && <Failed error={receipt.error} />}
      {receipt.isSuccess && <Confirmed receipt={receipt.data} />}
      <div className="mt-4 flex flex-wrap items-center gap-3">
        <ExternalButton href={explorer}>View in explorer</ExternalButton>
        <BackLink />
      </div>
    </Card>
  );
}

function Confirmed({ receipt }: { receipt: TransactionReceipt }) {

            <StatRow key={row.label} label={row.label}>
              {row.value}
            </StatRow>
          ))}
          <StatRow label="Gas used">{formatToken(receipt.gasUsed, 0, 0)}</StatRow>
          <StatRow label="Block">{receipt.blockNumber.toString()}</StatRow>
        </dl>
      ) : (

      )}
    </>
  );

function Failed({ error }: { error: Error }) {
```
